<script>
let eurRate = 0.92;
async function fetchEur() {
  try {
    const r = await fetch('https://api.coinbase.com/v2/exchange-rates?currency=USD');
    const d = await r.json();
    const rt = parseFloat(d?.data?.rates?.EUR);
    if (rt > 0) eurRate = rt;
  } catch(e) {}
}
fetchEur(); setInterval(fetchEur, 60000);

const toEur = v => v * eurRate;
const fE  = v => (v >= 0 ? '+' : '') + '€' + Math.abs(toEur(v)).toFixed(2);
const fEa = v => '€' + toEur(v).toFixed(2);

// ── KOMBINIERTER CHART ──────────────────────────────────────────
// Gelb=Heute(hell/nah), Orange=Woche, Bordeaux=Monat(dunkel/weit)
const C_DAY   = '#ffe566';
const C_WEEK  = '#e8943a';
const C_MONTH = '#8b1a2e';

const combCtx = document.getElementById('comboChart').getContext('2d');
const comboChart = new Chart(combCtx, {
  type: 'line',
  data: {
    labels: [],
    datasets: [
      { label:'Heute',  data:[], borderColor:C_DAY,   borderWidth:2,   pointRadius:0, fill:false, tension:0.4, yAxisID:'y'  },
      { label:'Woche',  data:[], borderColor:C_WEEK,  borderWidth:1.5, pointRadius:0, fill:false, tension:0.4, yAxisID:'y2' },
      { label:'Monat',  data:[], borderColor:C_MONTH, borderWidth:1.5, pointRadius:2, pointBackgroundColor:C_MONTH, fill:false, tension:0.3, yAxisID:'y3' }
    ]
  },
  options: {
    responsive:true, maintainAspectRatio:false, animation:false,
    interaction:{ mode:'index', intersect:false },
    plugins:{ legend:{display:false},
      tooltip:{ callbacks:{ label: c => c.dataset.label+': €'+toEur(c.parsed.y).toFixed(2) }}
    },
    scales:{
      x:{ ticks:{color:'#2a2050',font:{size:7},maxTicksLimit:6}, grid:{color:'rgba(26,31,56,.5)'} },
      y:{ display:false }, y2:{ display:false }, y3:{ display:false }
    }
  }
});
// Live-History für Heute-Kurve
const pH = [], pL = [];

function tick() {
  const n = new Date();
  document.getElementById('htime').textContent =
    String(n.getHours()).padStart(2,'0')+':'+String(n.getMinutes()).padStart(2,'0')+':'+String(n.getSeconds()).padStart(2,'0');
}
tick(); setInterval(tick,1000);

function render(d) {
  document.getElementById('hdot').style.background = 'var(--cyan)';
  document.getElementById('hdot').style.boxShadow  = '0 0 10px var(--cyan)';

  const total   = d.total || 0;
  const free    = d.usdc_free || 0;
  const margin  = d.margin_used || 0;
  const pnl     = d.today_pnl || 0;
  const wins    = d.wins || 0;
  const losses  = d.losses || 0;
  const lev     = d.leverage || 3;
  const profit  = d.total_profit || 0;
  const fees    = d.total_fees || 0;
  const wr      = (wins+losses) > 0 ? (wins/(wins+losses)*100).toFixed(0) : '--';

  document.getElementById('hlevBadge').textContent = lev + 'x';
  document.getElementById('hpf').textContent = '€' + toEur(total).toFixed(2);
  const hpnl = document.getElementById('hpnl');
  hpnl.textContent = fE(pnl);
  hpnl.style.color = pnl >= 0 ? 'var(--green)' : 'var(--red)';

  document.getElementById('hkPort').textContent = '€' + toEur(total).toFixed(2);
  document.getElementById('hkPortUsd').textContent = '$' + total.toFixed(2) + ' USDC';

  const hkp = document.getElementById('hkPnl');
  hkp.textContent = fE(pnl);
  hkp.style.color = pnl >= 0 ? 'var(--cyan)' : 'var(--red)';
  document.getElementById('hkPnlPct').textContent = total > 0 ? (pnl/total*100).toFixed(2)+'%' : '0%';

  const hkpr = document.getElementById('hkProfit');
  hkpr.textContent = '€' + toEur(profit).toFixed(2);
  hkpr.style.color = profit >= 0 ? 'var(--green)' : 'var(--red)';
  document.getElementById('hkFees').textContent = 'Fees: €' + toEur(fees).toFixed(4);

  document.getElementById('hkWr').textContent = wr + '%';
  document.getElementById('hkWr').style.color = wins >= losses ? 'var(--green)' : 'var(--red)';
  document.getElementById('hkWl').textContent = wins + 'W / ' + losses + 'L';

  document.getElementById('sUsd').textContent  = '€' + toEur(free).toFixed(2);
  document.getElementById('sUsd2').textContent = '$' + free.toFixed(2);
  document.getElementById('sMargin').textContent = '€' + toEur(margin).toFixed(2);
  document.getElementById('sPos').textContent = (d.positions||[]).length + '/3';

  const btc = d.btc_trend || 0;
  const btcEl = document.getElementById('sBtc');
  btcEl.textContent = btc > 0 ? '↑ UP' : btc < 0 ? '↓ DN' : '→ SIDE';
  btcEl.style.color = btc > 0 ? 'var(--green)' : btc < 0 ? 'var(--red)' : 'var(--gold)';
  document.getElementById('sFg').textContent = 'FG: ' + (d.fear_greed || '--');

  // ── CHART UPDATE ──────────────────────────────────────────────
  const n = new Date();
  const tl = String(n.getHours()).padStart(2,'0')+':'+String(n.getMinutes()).padStart(2,'0')+':'+String(n.getSeconds()).padStart(2,'0');

  // Heute-Kurve (Live RAM)
  pH.push(total); pL.push(tl);
  if (pH.length > 120) { pH.shift(); pL.shift(); }
  comboChart.data.labels          = pL;
  comboChart.data.datasets[0].data = pH;

  // Wochenkurve aus API
  const wc = d.week_chart || [], wl = d.week_chart_labels || [];
  if (wc.length > 1) {
    comboChart.data.datasets[1].data = wc;
    const wDiff = toEur(wc[wc.length-1] - wc[0]);
    const wdEl = document.getElementById('weekDiff');
    wdEl.textContent = (wDiff>=0?'+':'')+'€'+wDiff.toFixed(2);
    wdEl.style.color = wDiff>=0?'var(--green)':'var(--red)';
  }

  // Monatskurve aus API
  const mc = d.month_chart || [];
  if (mc.length > 1) {
    comboChart.data.datasets[2].data = mc;
    const mDiff = toEur(mc[mc.length-1] - mc[0]);
    const mdEl = document.getElementById('monthDiff');
    mdEl.textContent = (mDiff>=0?'+':'')+'€'+mDiff.toFixed(2);
    mdEl.style.color = mDiff>=0?'var(--green)':'var(--red)';
  }

  // Heute-Diff
  if (pH.length > 1) {
    const dDiff = toEur(pH[pH.length-1] - pH[0]);
    const ddEl = document.getElementById('dayDiff');
    ddEl.textContent = (dDiff>=0?'+':'')+'€'+dDiff.toFixed(2);
    ddEl.style.color = dDiff>=0?'var(--green)':'var(--red)';
  }

  comboChart.update('none');

  // Positionen
  const pos = d.positions || [];
  const pb = document.getElementById('posBody');
  if (!pos.length) {
    pb.innerHTML = '<div class="emp">Keine offenen Positionen</div>';
  } else {
    pb.innerHTML = pos.map(p => {
      const isLong = p.side === 'LONG';
      const col = p.pnl >= 0 ? 'var(--green)' : 'var(--red)';
      const bw = Math.min(Math.abs(p.pnl) / (lev * 6) * 100, 100).toFixed(0);
      const pair = (p.pair||'').replace('-PERP-INTX','');
      return `<div class="pos-item ${isLong?'long':'short'}">
        <div>
          <div class="pos-name">${pair}</div>
          <div class="side-badge ${isLong?'long':'short'}">${p.side}</div>
        </div>
        <div class="pos-mid">
          <div style="display:flex;align-items:baseline">
            <span class="pos-pnl" style="color:${col}">${(p.pnl>=0?'+':'')}${p.pnl.toFixed(1)}%</span>
            <span class="lev-tag">${lev}x</span>
          </div>
          <div class="pbar"><div class="pbarf" style="width:${bw}%;background:${col}"></div></div>
          <div style="display:flex;justify-content:space-between;margin-top:3px">
            <span style="font-size:8px;color:var(--t2)">Entry: €${toEur(p.entry).toFixed(4)}</span>
            <span style="font-size:8px;color:var(--t2)">Margin: €${toEur(p.margin||0).toFixed(2)}</span>
          </div>
        </div>
        <div style="text-align:right;font-size:8px">
          <div style="color:var(--red)">Stop: ${isLong ? '↓' : '↑'}${isLong?(((p.entry-p.stop)/p.entry*100).toFixed(1)):(((p.stop-p.entry)/p.entry*100).toFixed(1))}%</div>
          <div style="color:var(--green)">TP: ${isLong ? '↑' : '↓'}${isLong?(((p.tp-p.entry)/p.entry*100).toFixed(1)):(((p.entry-p.tp)/p.entry*100).toFixed(1))}%</div>
        </div>
      </div>`;
    }).join('');
  }

  // Trades
  const trades = d.trades || [];
  const tlEl = document.getElementById('tradeList');
  const closedTrades = trades.filter(t => t.action === 'CLOSE');
  if (!closedTrades.length) {
    tlEl.innerHTML = '<div class="emp">Noch keine geschlossenen Trades</div>';
  } else {
    tlEl.innerHTML = closedTrades.slice(0,30).map(t => {
      const isLong = t.side === 'LONG';
      const net = t.profit_usdc != null ? t.profit_usdc : 0;
      const pnl = t.pnl_pct != null ? t.pnl_pct : 0;
      const col = net >= 0 ? 'var(--green)' : 'var(--red)';
      const pair = (t.pair||'').replace('-PERP-INTX','').replace('-USDC','');
      return `<div class="ti">
        <div class="ts ${isLong?'lo':'sh'}">${isLong?'L':'S'}</div>
        <div style="flex:1;min-width:0">
          <div style="font-weight:700;font-size:10px">${pair}</div>
          <div style="font-size:8px;color:var(--t2)">${t.time||'--'} · ${t.reason||''}</div>
        </div>
        <div style="text-align:right;flex-shrink:0">
          <div style="font-family:'Syne',sans-serif;font-weight:700;font-size:11px;color:${col}">${net>=0?'+':''}€${toEur(net).toFixed(2)}</div>
          <div style="font-size:8px;color:var(--t2)">${pnl>=0?'+':''}${pnl.toFixed(1)}% · ${isLong?'LONG':'SHORT'}</div>
        </div>
      </div>`;
    }).join('');
  }

  // Alert
  const ab = document.getElementById('alertBox');
  const am = document.getElementById('alertMsg');
  if (d.circuit_breaker) { ab.className='al danger'; am.textContent='⚡ Circuit Breaker — 2h Pause'; }
  else if (d.daily_loss_hit) { ab.className='al warn'; am.textContent='⚠ Tages-Verlust-Limit erreicht'; }
  else if ((d.consec_losses||0) >= 2) { ab.className='al warn'; am.textContent='⚠ ' + d.consec_losses + ' Verluste in Folge'; }
  else { ab.className='al ok'; am.textContent='✓ Alle Systeme OK — Perp Bot aktiv'; }
}

function fetchData() {
  fetch('/api/status').then(r=>r.json()).then(d=>render(d)).catch(()=>{
    document.getElementById('hdot').style.background='var(--red)';
  });
}
fetchData(); setInterval(fetchData, 5000);
